{{- /* Cleans .Content for RSS: readers strip CSS and don't resolve relative URLs. */ -}} {{- $base := strings.TrimSuffix "/" site.BaseURL -}} {{- $c := .Content -}} {{- /* CSS-hidden on the site, a stray "#" in a reader. */ -}} {{- $c = replaceRE `\s*]*>#\s*` "" $c -}} {{- /* plumbing down to a plain ; also drops the base64 LQIP. */ -}} {{- $c = replaceRE `]*>` "" $c -}} {{- $c = replaceRE `\s+srcset="[^"]*"` "" $c -}} {{- $c = replaceRE `\s+sizes="[^"]*"` "" $c -}} {{- $c = replaceRE `\s+style="background-image:url\(data:image/[^"]*"` "" $c -}} {{- /* [^/] guard leaves //host URLs alone. */ -}} {{- $c = replaceRE `(src|href|poster)="/([^/])` (printf `${1}="%s/${2}` $base) $c -}} {{- return $c -}}
